THE CODE THAT RUNS ↓
gsap.registerPlugin(ScrollTrigger, ScrollSmoother);

// ScrollSmoother wraps #smooth-wrapper > #smooth-content and adds
// eased, inertia-based scrolling. data-speed on children = parallax.
ScrollSmoother.create({
  wrapper: "#smooth-wrapper",
  content: "#smooth-content",
  smooth: 1.5,        // seconds it takes to "catch up" to the real scroll
  effects: true       // enable data-speed / data-lag parallax attributes
});
